# The manifest: what the exe will accept, and the commit it fetches from.
set(assets_root ${PROJECT_SOURCE_DIR}/assets)
file(GLOB_RECURSE asset_files RELATIVE ${assets_root} CONFIGURE_DEPENDS ${assets_root}/*)
list(REMOVE_ITEM asset_files README.md)
list(SORT asset_files)
set(CRPP_ASSET_ENTRIES "")
foreach(asset IN LISTS asset_files)
  if(NOT asset MATCHES "^[A-Za-z0-9/_.-]+$")
    message(FATAL_ERROR "asset path needs URL escaping, which the fetcher does not do: ${asset}")
  endif()
  file(SHA256 ${assets_root}/${asset} asset_hash)
  file(SIZE ${assets_root}/${asset} asset_size)
  string(APPEND CRPP_ASSET_ENTRIES "    {\"${asset}\", ${asset_size}, \"${asset_hash}\"},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${assets_root}/${asset})
endforeach()
find_package(Git QUIET)
set(CRPP_ASSET_COMMIT "")
if(GIT_FOUND)
  execute_process(COMMAND ${GIT_EXECUTABLE} log -1 --format=%H -- assets WORKING_DIRECTORY ${PROJECT_SOURCE_DIR}
                  OUTPUT_VARIABLE CRPP_ASSET_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  execute_process(COMMAND ${GIT_EXECUTABLE} status --porcelain -- assets WORKING_DIRECTORY ${PROJECT_SOURCE_DIR}
                  OUTPUT_VARIABLE assets_dirty OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(assets_dirty)
    message(WARNING "assets/ has uncommitted changes, so this build's manifest matches no commit")
  endif()
endif()
