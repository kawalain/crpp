# The manifest: what the exe will accept, and where it fetches from. Everything goes
# through jsDelivr, pinned to commits, whose answers it may cache for good.
set(CRPP_TWEMOJI_COMMIT b6b55fef1e8636b540a6d016a4729ca8cdf2e60b)     # v17.0.3
# By tag: jsDelivr answers 403 for this large repository by commit. The hash still pins it.
set(CRPP_PRETENDARD_REF v1.3.9)
set(assets_root ${PROJECT_SOURCE_DIR}/assets)
file(GLOB_RECURSE asset_files RELATIVE ${assets_root} CONFIGURE_DEPENDS ${assets_root}/*)
list(REMOVE_ITEM asset_files README.md)
list(SORT asset_files)
set(CRPP_ASSET_ENTRIES "")
set(picker_ids "")
set(custom_ids "")
foreach(asset IN LISTS asset_files)
  if(NOT asset MATCHES "^[a-z0-9/_.A-Z-]+$")
    message(FATAL_ERROR "asset path needs URL escaping, which the fetcher does not do: ${asset}")
  endif()
  file(SHA256 ${assets_root}/${asset} asset_hash)
  file(SIZE ${assets_root}/${asset} asset_size)
  string(APPEND CRPP_ASSET_ENTRIES "    {\"${asset}\", ${asset_size}, \"${asset_hash}\", 0},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${assets_root}/${asset})
  if(asset MATCHES "^emoji/([0-9][0-9])\\.svg$")
    list(APPEND custom_ids ${CMAKE_MATCH_1})
  elseif(asset MATCHES "^emoji/(.+)\\.svg$")
    list(APPEND picker_ids ${CMAKE_MATCH_1})
  endif()
endforeach()

# Third-party files: the cache keeps them out of the repository for a development build,
# which runs from a folder next to the exe.
option(CRPP_FETCH_DEV_ASSETS "Fetch the third-party assets for running this build from its own folder" ON)
set(CRPP_ASSET_CACHE ${PROJECT_SOURCE_DIR}/.cache/remote-assets CACHE PATH "Where a development build keeps them")
set(lock_file ${PROJECT_SOURCE_DIR}/cmake/remote_assets.txt)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${lock_file})
file(STRINGS ${lock_file} lock_lines REGEX "^[^#]")
set(fetch_config "")
set(fetched_paths "")
foreach(line IN LISTS lock_lines)
  string(REPLACE " " ";" fields "${line}")
  list(GET fields 0 lock_source)
  list(GET fields 1 lock_path)
  list(GET fields 2 lock_size)
  list(GET fields 3 lock_hash)
  list(LENGTH fields field_count)
  if(lock_source STREQUAL "twemoji")
    set(source_index 1)
    set(lock_url "jdecked/twemoji@${CRPP_TWEMOJI_COMMIT}/assets/svg/")
  elseif(lock_source STREQUAL "pretendard")
    set(source_index 2)
    set(lock_url "orioncactus/pretendard@${CRPP_PRETENDARD_REF}/packages/pretendard-gov/dist/public/variable/")
  else()
    message(FATAL_ERROR "unknown source in ${lock_file}: ${lock_source}")
  endif()
  string(APPEND CRPP_ASSET_ENTRIES "    {\"${lock_path}\", ${lock_size}, \"${lock_hash}\", ${source_index}},\n")
  if(lock_path MATCHES "^emoji/(.+)\\.svg$" AND field_count EQUAL 4)
    list(APPEND picker_ids ${CMAKE_MATCH_1})
  endif()
  if(WIN32 AND CRPP_FETCH_DEV_ASSETS)
    set(cached ${CRPP_ASSET_CACHE}/${lock_path})
    set(have_size 0)
    if(EXISTS ${cached})
      file(SIZE ${cached} have_size)
    endif()
    if(NOT have_size EQUAL lock_size)
      cmake_path(GET lock_path FILENAME lock_name)
      string(APPEND fetch_config "url = \"https://cdn.jsdelivr.net/gh/${lock_url}${lock_name}\"\noutput = \"${cached}\"\n")
      list(APPEND fetched_paths "${lock_path}=${lock_hash}")
    endif()
  endif()
endforeach()
# curl, which Windows has, because file(DOWNLOAD) cannot verify certificates in every
# CMake build. A failure leaves a build that uses the system's font and emoji.
if(fetch_config)
  find_program(CURL_PROGRAM curl)
  set(fetch_file ${CMAKE_CURRENT_BINARY_DIR}/fetch_assets.txt)
  file(WRITE ${fetch_file} "${fetch_config}")
  file(MAKE_DIRECTORY ${CRPP_ASSET_CACHE}/fonts ${CRPP_ASSET_CACHE}/emoji)
  set(fetch_result "curl was not found")
  if(CURL_PROGRAM)
    execute_process(COMMAND ${CURL_PROGRAM} --fail --silent --show-error --location --max-time 120
                            --parallel --parallel-max 8 --config ${fetch_file} RESULT_VARIABLE fetch_result ERROR_QUIET)
  endif()
  if(NOT fetch_result EQUAL 0)
    message(WARNING "Could not fetch the third-party assets (${fetch_result}). This build uses the system's font and emoji.")
  endif()
  foreach(item IN LISTS fetched_paths)
    string(REPLACE "=" ";" parts "${item}")
    list(GET parts 0 fetched_path)
    list(GET parts 1 expected_hash)
    set(have_hash "")
    if(EXISTS ${CRPP_ASSET_CACHE}/${fetched_path})
      file(SHA256 ${CRPP_ASSET_CACHE}/${fetched_path} have_hash)
    endif()
    if(NOT have_hash STREQUAL expected_hash)
      file(REMOVE ${CRPP_ASSET_CACHE}/${fetched_path})
    endif()
  endforeach()
endif()
