function(crpp_sound_test name)
  add_executable(crpp_sound_${name}_test tests/${name}_test.cpp)
  target_link_libraries(crpp_sound_${name}_test PRIVATE crpp_sound ${ARGN})
  target_include_directories(crpp_sound_${name}_test PRIVATE ${PROJECT_SOURCE_DIR}/tests/support)
  target_compile_definitions(crpp_sound_${name}_test PRIVATE CRPP_SOUNDS_DIR="${PROJECT_SOURCE_DIR}/assets/sounds")
  crpp_target_defaults(crpp_sound_${name}_test)
  add_test(NAME sound.${name} COMMAND crpp_sound_${name}_test)
endfunction()
