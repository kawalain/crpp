# Third-party files: a development build runs from a folder next to the exe, and the cache
# keeps them out of the repository. They come from their sources in the list.
option(CRPP_FETCH_DEV_ASSETS "Fetch the third-party assets for running this build from its own folder" ON)
set(CRPP_ASSET_CACHE ${PROJECT_SOURCE_DIR}/.cache/remote-assets CACHE PATH "Where a development build keeps them")
set(CRPP_REMOTE_ASSETS ${PROJECT_SOURCE_DIR}/cmake/remote_assets.txt)
set(CRPP_REMOTE_ASSETS ${CRPP_REMOTE_ASSETS} PARENT_SCOPE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CRPP_REMOTE_ASSETS})
if(WIN32 AND CRPP_FETCH_DEV_ASSETS)
  file(STRINGS ${CRPP_REMOTE_ASSETS} lock_lines REGEX "^[^#]")
  set(fetch_config "")
  set(fetched_paths "")
  foreach(line IN LISTS lock_lines)
    string(REPLACE " " ";" fields "${line}")
    list(GET fields 0 lock_source)
    list(GET fields 1 lock_path)
    if(lock_source STREQUAL "source")
      list(GET fields 2 source_url_${lock_path})
      continue()
    endif()
    list(GET fields 2 lock_size)
    list(GET fields 3 lock_hash)
    cmake_path(GET lock_path FILENAME lock_name)
    if(lock_source MATCHES "^([^:]+):(.+)$")
      set(lock_source ${CMAKE_MATCH_1})
      set(lock_name ${CMAKE_MATCH_2})
    endif()
    if(NOT DEFINED source_url_${lock_source})
      message(FATAL_ERROR "no source line for ${lock_source} in ${CRPP_REMOTE_ASSETS}")
    endif()
    set(cached ${CRPP_ASSET_CACHE}/${lock_path})
    set(have_size 0)
    if(EXISTS ${cached})
      file(SIZE ${cached} have_size)
    endif()
    if(NOT have_size EQUAL lock_size)
      string(APPEND fetch_config "url = \"${source_url_${lock_source}}${lock_name}\"\noutput = \"${cached}\"\n")
      list(APPEND fetched_paths "${lock_path}=${lock_hash}")
    endif()
  endforeach()
  # curl, which Windows has, because file(DOWNLOAD) cannot verify certificates in every
  # CMake build. A failure leaves a build that uses the system's font and emoji.
  if(fetch_config)
    find_program(CURL_PROGRAM curl)
    set(fetch_file ${CMAKE_CURRENT_BINARY_DIR}/fetch_assets.txt)
    file(WRITE ${fetch_file} "${fetch_config}")
    file(MAKE_DIRECTORY ${CRPP_ASSET_CACHE}/fonts ${CRPP_ASSET_CACHE}/emoji ${CRPP_ASSET_CACHE}/licenses)
    set(fetch_result "curl was not found")
    if(CURL_PROGRAM)
      execute_process(COMMAND ${CURL_PROGRAM} --fail --silent --show-error --location --max-time 120
                              --parallel --parallel-max 8 --config ${fetch_file} RESULT_VARIABLE fetch_result ERROR_QUIET)
    endif()
    if(NOT fetch_result EQUAL 0)
      message(WARNING "Could not fetch the third-party assets (${fetch_result}). This build uses the system's font and emoji.")
    endif()
    foreach(item IN LISTS fetched_paths)
      string(REPLACE "=" ";" parts "${item}")
      list(GET parts 0 fetched_path)
      list(GET parts 1 expected_hash)
      set(have_hash "")
      if(EXISTS ${CRPP_ASSET_CACHE}/${fetched_path})
        file(SHA256 ${CRPP_ASSET_CACHE}/${fetched_path} have_hash)
      endif()
      if(NOT have_hash STREQUAL expected_hash)
        file(REMOVE ${CRPP_ASSET_CACHE}/${fetched_path})
      endif()
    endforeach()
  endif()
endif()
